The Game of Life

Conway's cells, born and dying by how many neighbours they have, drawn again five times a second.

Run in the playground ↗ game-of-life.moth
The code, 56 lines 
# The Game of Life
# Conway's cells, born and dying by how many neighbours they have, drawn again five times a second.

set size to 24
set cell to 18
set alive to empty dictionary # "x,y" for each living cell

to key_of with x, y
    return x + "," + y
end

to neighbours with x, y
    set count to 0
    for each dx from -1 to 1
        for each dy from -1 to 1
            if not (dx is 0 and dy is 0) and key_of(x + dx, y + dy) is in alive
                add 1 to count
            end
        end
    end
    return count
end

to step
    set next to empty dictionary
    for each x from 0 to size - 1
        for each y from 0 to size - 1
            set n to neighbours(x, y)
            if n is 3 or (n is 2 and key_of(x, y) is in alive)
                set item key_of(x, y) of next to yes
            end
        end
    end
    set alive to next
end

to show
    clear
    pen color "#14100c"
    draw box at -size * cell / 2, -size * cell / 2 size size * cell by size * cell
    pen color "#f4b54a"
    for each x from 0 to size - 1
        for each y from 0 to size - 1
            if key_of(x, y) is in alive
                draw box at (x - size / 2) * cell + 1, (y - size / 2) * cell + 1 size cell - 2 by cell - 2
            end
        end
    end
end

# a glider, and a row of ten that grows into something else
for each place in list of (list of 1, 20), (list of 2, 19), (list of 0, 18), (list of 1, 18), (list of 2, 18)
    set item key_of(item 1 of place, item 2 of place) of alive to yes
end
for each x from 7 to 16
    set item key_of(x, 8) of alive to yes
end
show()

every 200 milliseconds
    step()
    show()
end

# Bouncing balls
# Balls that fall, bounce off the floor and the walls, and lose a little each bounce, drawn 50 times a second.
# Click to drop another.

set balls to empty list
set colours to list of "#f4b54a", "#ff7a2f", "#7fc8d8", "#c3d98a", "#e87ba4"

to drop with x, y
    add (record with x x, y y, dx (random number from -150 to 150), dy 0, colour item (random number from 1 to 5) of colours) to balls
end

to fall with seconds
    for each ball in balls
        set dy of ball to dy of ball - 600 * seconds # gravity, pulling down
        set x of ball to x of ball + dx of ball * seconds
        set y of ball to y of ball + dy of ball * seconds
        if y of ball is less than -230
            set y of ball to -230
            set dy of ball to 0 - dy of ball * 0.8
        end
        if x of ball is less than -230 or x of ball is greater than 230
            set dx of ball to 0 - dx of ball
            set x of ball to largest of (list of -230, smallest of (list of x of ball, 230))
        end
    end
end

to show
    clear
    for each ball in balls
        pen color colour of ball
        draw circle at x of ball, y of ball size 20
    end
    pen color "#8a7a63"
    draw "click to drop a ball" at -80, 225
end

drop(-100, 200)
drop(0, 150)
drop(120, 220)
show()

every 20 milliseconds
    fall(0.02)
    show()
end

when the mouse is clicked with x and y
    drop(x, y)
end

# The Mandelbrot set
# Each point, a complex number c: start at 0, square and add c, again and again. Those that stay small are in
# the set, drawn here in letters, the denser the longer they took to leave.

set shades to split " .:-=+*#%@" by ""
for each row from 0 to 23
    set y to 1.2 - row * 0.1
    set line to ""
    for each column from 0 to 69
        set x to -2.1 + column * 0.04
        set a to 0
        set b to 0
        set steps to 0
        repeat while steps is less than 60 and a * a + b * b is at most 4
            set next_a to a * a - b * b + x
            set b to 2 * a * b + y
            set a to next_a
            add 1 to steps
        end
        if steps is 60
            add "@" to line
        else
            add item (remainder of steps divided by 9 + 1) of shades to line
        end
    end
    say line
end

# A class's results
# Marks summed up with stats.moth (mean, median, spread, the top and bottom quarter), each student's grade,
# and a chart of how many got each grade.

use "stats.moth" as stats

set marks to dictionary where "Ada" is 91, "Ben" is 67, "Cai" is 78, "Dee" is 84, "Eli" is 55, "Fay" is 72,
    "Gus" is 88, "Hal" is 61, "Ivy" is 95, "Jo" is 70, "Kit" is 79, "Lu" is 83

set all to empty list
for each name in keys of marks
    add item name of marks to all

    if mark is at least 85
        return "A"
    else if mark is at least 75
        return "B"
    else if mark is at least 65
        return "C"
    end
    return "D"
end

for each name in keys of marks
    set grade to grade_of(item name of marks)
    set item grade of grades to item grade of grades + 1
    if grade is "A"
        say name " got an A, with " item name of marks
    end
end

# The Towers of Hanoi
# Move a tower of discs from one peg to another, a disc at a time, never a big one on a small one: a function
# that calls itself does it in 2^n - 1 moves.

set moves to 0

to move_tower with discs, from, to_peg, spare, telling
    if discs is 0
        return nothing
    end
    move_tower(discs - 1, from, spare, to_peg, telling)
    add 1 to moves
    if telling
        say "Move disc " discs " from " from " to " to_peg
    end
    move_tower(discs - 1, spare, to_peg, from, telling)
end

move_tower(3, "left", "right", "middle", yes)
say moves " moves for 3 discs"

for each discs in list of 10, 16
    set moves to 0
    move_tower(discs, "left", "right", "middle", no)
    say moves " moves for " discs " discs: 2 to the power of " discs ", less 1"
end

Run in the playground ↗ quicksort.moth
The code, 37 lines 
# Quicksort
# Sorting by picking a pivot and sorting what's smaller and what's bigger, each the same way: `give` adds to
# the list a function gives back.

to smaller_than with things, pivot
    for each x in things
        if x is less than pivot
            give x
        end
    end
end

to not_smaller_than with things, pivot
    for each x in things
        if x is at least pivot
            give x
        end
    end
end

to quicksort with things
    if length of things is at most 1
        return things
    end
    set pivot to item 1 of things
    set rest to items 2 to (length of things) of things
    set sorted to quicksort(smaller_than(rest, pivot))
    add pivot to sorted
    for each x in quicksort(not_smaller_than(rest, pivot))
        add x to sorted
    end
    return sorted
end

say quicksort(list of 38, 27, 43, 3, 9, 82, 10)
say quicksort(split "the quick brown fox jumps over the lazy dog" by " ")

set many to empty list
repeat 2000 times
    add random number from 1 to 100000 to many
end
say "2,000 random numbers sorted the same as Moth sorts them: " quicksort(many) is many in order

Run in the playground ↗ roman-numerals.moth
The code, 29 lines 
# Roman numerals
# A number written as the Romans wrote it, and read back: the biggest symbol that fits, again and again.

set values to list of 1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1
set symbols to list of "M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"

to roman with n
    set made to ""
    for each k from 1 to length of values
        repeat while n is at least item k of values
            add item k of symbols to made
            set n to n - item k of values
        end
    end
    return made
end

to number_of_roman with written
    set n to 0
    set rest to written
    for each k from 1 to length of values
        set symbol to item k of symbols
        repeat while length of rest is at least length of symbol and position of symbol in rest is 1
            set n to n + item k of values
            set rest to items (length of symbol + 1) to (length of rest) of rest
        end
    end
    return n
end

for each year in list of 1999, 2026, 3888
    say year " is " roman(year) ", and " roman(year) " is " number_of_roman(roman(year))
end

A Caesar cipher

Each letter moved along the alphabet, as Julius Caesar hid his letters, and moved back again.

Phhw ph dw wkh rog eulgjh, dw qrrq.
Meet me at the old bridge, at noon.
Broken: the shift was 3: Meet me at the old bridge, at noon.
Run in the playground ↗ caesar-cipher.moth
The code, 37 lines 
# A Caesar cipher
# Each letter moved along the alphabet, as Julius Caesar hid his letters, and moved back again.

set alphabet to "abcdefghijklmnopqrstuvwxyz"

to shifted with message, by
    set made to ""
    for each letter in split message by ""
        set place to position of (lowercase of letter) in alphabet
        if place is 0
            add letter to made # not a letter: as it is
        else
            set moved to item (remainder of (place - 1 + by + 26) divided by 26 + 1) of alphabet
            if letter is not lowercase of letter
                set moved to uppercase of moved
            end
            add moved to made
        end
    end
    return made
end

set secret to shifted("Meet me at the old bridge, at noon.", 3)
say secret
say shifted(secret, -3)

# breaking it: try every shift, and keep the one with the most common English letters
set best to 0
set best_score to -1
for each guess from 0 to 25
    set tried to shifted(secret, -guess)
    set score to 0
    for each letter in split "etaoin" by ""
        set score to score + (length of (split tried by letter)) - 1
    end
    if score is greater than best_score
        set best to guess
        set best_score to score
    end
end
say "Broken: the shift was " best ": " shifted(secret, -best)

# Counting words
# Which words a text uses most: its words found with a pattern, counted, and the most common first.

use "text.moth" as text
use "collections.moth" as collections

set opening to "It was the best of times, it was the worst of times, it was the age of wisdom, it was the age of foolishness, it was the epoch of belief, it was the epoch of incredulity."

The code, 35 lines 
# A bank account
# A kind: a record with functions of its own, made by `account(...)`, whose deposit and withdraw keep the
# balance right, and say no to taking out more than there is.

kind account with owner, balance or 0
    to deposit with self, amount
        set balance of self to balance of self + amount
    end
    to withdraw with self, amount
        if amount is greater than balance of self
            fail with owner of self + " has only " + balance of self + ", not " + amount
        end
        set balance of self to balance of self - amount
    end
    to shown with self
        return owner of self + "'s account: " + balance of self
    end
end

set sam to account("Sam")
set ana to account("Ana", 500)
sam's deposit(120)
ana's withdraw(75)
say sam
say ana

try
    sam's withdraw(1000)
if it fails with why
    say "No: " why
end

to move_money with from, to_account, amount
    from's withdraw(amount)
    to_account's deposit(amount)
end
move_money(ana, sam, 200)
say sam
say ana

# A month's calendar
# The days of a month in weeks, Monday first, with dates.moth: what day the 1st is, how many days the month
# has, and a date written as people write it.

use "dates.moth" as dates

set year to 2026
set month to 11
set first to dates's date(year, month, 1)
say "      " dates's format_date(first, "MMMM YYYY")
say "Mo Tu We Th Fr Sa Su"

set line to ""
for each blank from 2 to dates's weekday_number(first)
    add "   " to line
end
for each day from 1 to dates's days_in_month(year, month)
    if day is less than 10
        add " " to line
    end
    add day + " " to line
    if dates's weekday_number(dates's date(year, month, day)) is 7
        say line
        set line to ""
    end
end
if line is not ""
    say line
end

say "Christmas " year " is on a " dates's weekday_name(christmas) ", " dates's days_between(first, christmas) " days after " dates's format_date(first, "D MMMM")
